104 inline state::ID output_state_for(const std::optional<args> &args) noexcept
105 {
106 if (const auto parsed = args ? args->get() : std::nullopt; parsed.has_value())
107 {
108 if (const auto it = parsed->find(mazes::args::OUTPUT_ID_WORD_STR); it != parsed->cend())
109 {
110 const std::string output = it->second;
111 if (output.empty() || output == STDOUT_FORMAT_STR)
112 {
113 return state::ID::STRINGIFYING;
114 }
115
116 const auto ext = std::filesystem::path{output}.extension().string();
117 if (ext.empty())
118 {
119 return state::ID::STRINGIFYING;
120 }
121
122 auto normalized = ext;
123 if (!normalized.empty() && normalized.front() == '.')
124 {
125 normalized.erase(normalized.begin());
126 }
127
128 try
129 {
130 switch (to_output_format_from_sv(normalized))
131 {
132 case output_format::PLAIN_TEXT:
133 case output_format::STDOUT:
134 return state::ID::STRINGIFYING;
135 case output_format::JSON_FILE:
136 return state::ID::PARSING;
137 case output_format::WAVEFRONT_OBJECT_FILE:
138 return state::ID::WAVEFRONT_OBJECTIFYING;
139 default:
140 break;
141 }
142 }
143 catch (...)
144 {
145 }
146
147 if (normalized == "png" || normalized == "jpg" || normalized == "jpeg" || normalized == "bmp")
148 {
149 return state::ID::PIXELIZING;
150 }
151 }
152 }
153
154 return state::ID::STRINGIFYING;
